23    where

27        let beta = E::ScalarField::rand(rng);

29        let delta = E::ScalarField::rand(rng);

31        let g1_generator = E::G1::rand(rng);
32        let g2_generator = E::G2::rand(rng);

34        Self::generate_parameters_with_qap(

46    /// Create parameters for a circuit, given some toxic waste, R1CS to QAP calculator and group generators
47    pub fn generate_parameters_with_qap<C>(
48        circuit: C,
49        alpha: E::ScalarField,
50        beta: E::ScalarField,
51        gamma: E::ScalarField,
52        delta: E::ScalarField,
53        g1_generator: E::G1,
54        g2_generator: E::G2,
55        rng: &mut impl Rng,
56    ) -> R1CSResult<ProvingKey<E>>
57    where
58        C: ConstraintSynthesizer<E::ScalarField>,
59    {
60        type D<F> = GeneralEvaluationDomain<F>;
61
62        let setup_time = start_timer!(|| "Groth16::Generator");
63        let cs = ConstraintSystem::new_ref();
64        cs.set_optimization_goal(OptimizationGoal::Constraints);
65        cs.set_mode(SynthesisMode::Setup);
66
67        // Synthesize the circuit.
68        let synthesis_time = start_timer!(|| "Constraint synthesis");
69        circuit.generate_constraints(cs.clone())?;
70        end_timer!(synthesis_time);
71
72        let lc_time = start_timer!(|| "Inlining LCs");
73        cs.finalize();
74        end_timer!(lc_time);
75
76        // Following is the mapping of symbols from the Groth16 paper to this implementation
77        // l -> num_instance_variables
78        // m -> qap_num_variables
79        // x -> t
80        // t(x) - zt
81        // u_i(x) -> a
82        // v_i(x) -> b
83        // w_i(x) -> c
84
85        ///////////////////////////////////////////////////////////////////////////
86        let domain_time = start_timer!(|| "Constructing evaluation domain");
87
88        let domain_size = cs.num_constraints() + cs.num_instance_variables();
89        let domain = D::new(domain_size).ok_or(SynthesisError::PolynomialDegreeTooLarge)?;
90        let t = domain.sample_element_outside_domain(rng);
91
92        end_timer!(domain_time);
93        ///////////////////////////////////////////////////////////////////////////
94
95        let reduction_time = start_timer!(|| "R1CS to QAP Instance Map with Evaluation");
96        let num_instance_variables = cs.num_instance_variables();
97        let (a, b, c, zt, qap_num_variables, m_raw) =
98            QAP::instance_map_with_evaluation::<E::ScalarField, D<E::ScalarField>>(cs, &t)?;
99        end_timer!(reduction_time);
100
101        // Compute query densities
102        let non_zero_a: usize = cfg_into_iter!(0..qap_num_variables)
103            .map(|i| usize::from(!a[i].is_zero()))
104            .sum();
105
106        let non_zero_b: usize = cfg_into_iter!(0..qap_num_variables)
107            .map(|i| usize::from(!b[i].is_zero()))
108            .sum();
109
110        let scalar_bits = E::ScalarField::MODULUS_BIT_SIZE as usize;
111
112        let gamma_inverse = gamma.inverse().ok_or(SynthesisError::UnexpectedIdentity)?;
113        let delta_inverse = delta.inverse().ok_or(SynthesisError::UnexpectedIdentity)?;
114
115        let gamma_abc = cfg_iter!(a[..num_instance_variables])
116            .zip(&b[..num_instance_variables])
117            .zip(&c[..num_instance_variables])
118            .map(|((a, b), c)| (beta * a + &(alpha * b) + c) * &gamma_inverse)
119            .collect::<Vec<_>>();
120
121        let l = cfg_iter!(a[num_instance_variables..])
122            .zip(&b[num_instance_variables..])
123            .zip(&c[num_instance_variables..])
124            .map(|((a, b), c)| (beta * a + &(alpha * b) + c) * &delta_inverse)
125            .collect::<Vec<_>>();
126
127        drop(c);
128
129        // Compute B window table
130        let g2_time = start_timer!(|| "Compute G2 table");
131        let g2_window = FixedBase::get_mul_window_size(non_zero_b);
132        let g2_table = FixedBase::get_window_table::<E::G2>(scalar_bits, g2_window, g2_generator);
133        end_timer!(g2_time);
134
135        // Compute the B-query in G2
136        let b_g2_time = start_timer!(|| "Calculate B G2");
137        let b_g2_query = FixedBase::msm::<E::G2>(scalar_bits, g2_window, &g2_table, &b);
138        drop(g2_table);
139        end_timer!(b_g2_time);
140
141        // Compute G window table
142        let g1_window_time = start_timer!(|| "Compute G1 window table");
143        let g1_window =
144            FixedBase::get_mul_window_size(non_zero_a + non_zero_b + qap_num_variables + m_raw + 1);
145        let g1_table = FixedBase::get_window_table::<E::G1>(scalar_bits, g1_window, g1_generator);
146        end_timer!(g1_window_time);
147
148        // Generate the R1CS proving key
149        let proving_key_time = start_timer!(|| "Generate the R1CS proving key");
150
151        let alpha_g1 = g1_generator.mul_bigint(&alpha.into_bigint());
152        let beta_g1 = g1_generator.mul_bigint(&beta.into_bigint());
153        let beta_g2 = g2_generator.mul_bigint(&beta.into_bigint());
154        let delta_g1 = g1_generator.mul_bigint(&delta.into_bigint());
155        let delta_g2 = g2_generator.mul_bigint(&delta.into_bigint());
156
157        // Compute the A-query
158        let a_time = start_timer!(|| "Calculate A");
159        let a_query = FixedBase::msm::<E::G1>(scalar_bits, g1_window, &g1_table, &a);
160        drop(a);
161        end_timer!(a_time);
162
163        // Compute the B-query in G1
164        let b_g1_time = start_timer!(|| "Calculate B G1");
165        let b_g1_query = FixedBase::msm::<E::G1>(scalar_bits, g1_window, &g1_table, &b);
166        drop(b);
167        end_timer!(b_g1_time);
168
169        // Compute the H-query
170        let h_time = start_timer!(|| "Calculate H");
171        let h_query = FixedBase::msm::<E::G1>(
172            scalar_bits,
173            g1_window,
174            &g1_table,
175            &QAP::h_query_scalars::<_, D<E::ScalarField>>(m_raw - 1, t, zt, delta_inverse)?,
176        );
177
178        end_timer!(h_time);
179
180        // Compute the L-query
181        let l_time = start_timer!(|| "Calculate L");
182        let l_query = FixedBase::msm::<E::G1>(scalar_bits, g1_window, &g1_table, &l);
183        drop(l);
184        end_timer!(l_time);
185
186        end_timer!(proving_key_time);
187
188        // Generate R1CS verification key
189        let verifying_key_time = start_timer!(|| "Generate the R1CS verification key");
190        let gamma_g2 = g2_generator.mul_bigint(&gamma.into_bigint());
191        let gamma_abc_g1 = FixedBase::msm::<E::G1>(scalar_bits, g1_window, &g1_table, &gamma_abc);
192
193        drop(g1_table);
194
195        end_timer!(verifying_key_time);
196
197        let vk = VerifyingKey::<E> {
198            alpha_g1: alpha_g1.into_affine(),
199            beta_g2: beta_g2.into_affine(),
200            gamma_g2: gamma_g2.into_affine(),
201            delta_g2: delta_g2.into_affine(),
202            gamma_abc_g1: E::G1::normalize_batch(&gamma_abc_g1),
203        };
204
205        let batch_normalization_time = start_timer!(|| "Convert proving key elements to affine");
206        let a_query = E::G1::normalize_batch(&a_query);
207        let b_g1_query = E::G1::normalize_batch(&b_g1_query);
208        let b_g2_query = E::G2::normalize_batch(&b_g2_query);
209        let h_query = E::G1::normalize_batch(&h_query);
210        let l_query = E::G1::normalize_batch(&l_query);
211        end_timer!(batch_normalization_time);
212        end_timer!(setup_time);
213
214        Ok(ProvingKey {
215            vk,
216            beta_g1: beta_g1.into_affine(),
217            delta_g1: delta_g1.into_affine(),
218            a_query,
219            b_g1_query,
220            b_g2_query,
221            h_query,
222            l_query,
223        })
224    }
