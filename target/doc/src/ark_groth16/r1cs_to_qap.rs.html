15/// Computes the inner product of `terms` with `assignment`.
16pub fn evaluate_constraint<'a, LHS, RHS, R>(terms: &'a [(LHS, usize)], assignment: &'a [RHS]) -> R
17where
18    LHS: One + Send + Sync + PartialEq,
19    RHS: Send + Sync + core::ops::Mul<&'a LHS, Output = RHS> + Copy,
20    R: Zero + Send + Sync + AddAssign<RHS> + core::iter::Sum,
21{
22    // Need to wrap in a closure when using Rayon
23    #[cfg(feature = "parallel")]
24    let zero = || R::zero();
25    #[cfg(not(feature = "parallel"))]
26    let zero = R::zero();
27
28    let res = cfg_iter!(terms).fold(zero, |mut sum, (coeff, index)| {
29        let val = &assignment[*index];
30
31        if coeff.is_one() {
32            sum += *val;
33        } else {
34            sum += val.mul(coeff);
35        }
36
37        sum
38    });
39
40    // Need to explicitly call `.sum()` when using Rayon
41    #[cfg(feature = "parallel")]
42    return res.sum();
43    #[cfg(not(feature = "parallel"))]
44    return res;
45}

47/// Computes instance and witness reductions from R1CS to
48/// Quadratic Arithmetic Programs (QAPs).
49pub trait R1CSToQAP {
50    /// Computes a QAP instance corresponding to the R1CS instance defined by `cs`.
51    fn instance_map_with_evaluation<F: PrimeField, D: EvaluationDomain<F>>(
52        cs: ConstraintSystemRef<F>,
53        t: &F,
54    ) -> Result<(Vec<F>, Vec<F>, Vec<F>, F, usize, usize), SynthesisError>;
55
56    #[inline]
57    /// Computes a QAP witness corresponding to the R1CS witness defined by `cs`.
58    fn witness_map<F: PrimeField, D: EvaluationDomain<F>>(
59        prover: ConstraintSystemRef<F>,
60    ) -> Result<Vec<F>, SynthesisError> {
61        let matrices = prover.to_matrices().unwrap();
62        let num_inputs = prover.num_instance_variables();
63        let num_constraints = prover.num_constraints();
64
65        let cs = prover.borrow().unwrap();
66        let prover = cs.deref();
67
68        let full_assignment = [
69            prover.instance_assignment.as_slice(),
70            prover.witness_assignment.as_slice(),
71        ]
72        .concat();
73
74        Self::witness_map_from_matrices::<F, D>(
75            &matrices,
76            num_inputs,
77            num_constraints,
78            &full_assignment,
79        )
80    }
81
82    /// Computes a QAP witness corresponding to the R1CS witness defined by `cs`.
83    fn witness_map_from_matrices<F: PrimeField, D: EvaluationDomain<F>>(
84        matrices: &ConstraintMatrices<F>,
85        num_inputs: usize,
86        num_constraints: usize,
87        full_assignment: &[F],
88    ) -> R1CSResult<Vec<F>>;
89
90    /// Computes the exponents that the generator uses to calculate base
91    /// elements which the prover later uses to compute `h(x)t(x)/delta`.
92    fn h_query_scalars<F: PrimeField, D: EvaluationDomain<F>>(
93        max_power: usize,
94        t: F,
95        zt: F,
96        delta_inverse: F,
97    ) -> Result<Vec<F>, SynthesisError>;
98}

100/// Computes the R1CS-to-QAP reduction defined in [`libsnark`](https://github.com/scipr-lab/libsnark/blob/2af440246fa2c3d0b1b0a425fb6abd8cc8b9c54d/libsnark/reductions/r1cs_to_qap/r1cs_to_qap.tcc).
101pub struct LibsnarkReduction;
102
103impl R1CSToQAP for LibsnarkReduction {
104    #[inline]
105    #[allow(clippy::type_complexity)]
106    fn instance_map_with_evaluation<F: PrimeField, D: EvaluationDomain<F>>(
107        cs: ConstraintSystemRef<F>,
108        t: &F,
109    ) -> R1CSResult<(Vec<F>, Vec<F>, Vec<F>, F, usize, usize)> {
110        let matrices = cs.to_matrices().unwrap();
111        let domain_size = cs.num_constraints() + cs.num_instance_variables();
112        let domain = D::new(domain_size).ok_or(SynthesisError::PolynomialDegreeTooLarge)?;
113        let domain_size = domain.size();
114
115        let zt = domain.evaluate_vanishing_polynomial(*t);
116
117        // Evaluate all Lagrange polynomials
118        let coefficients_time = start_timer!(|| "Evaluate Lagrange coefficients");
119        let u = domain.evaluate_all_lagrange_coefficients(*t);
120        end_timer!(coefficients_time);
121
122        let qap_num_variables = (cs.num_instance_variables() - 1) + cs.num_witness_variables();
123
124        let mut a = vec![F::zero(); qap_num_variables + 1];
125        let mut b = vec![F::zero(); qap_num_variables + 1];
126        let mut c = vec![F::zero(); qap_num_variables + 1];
127
128        {
129            let start = 0;
130            let end = cs.num_instance_variables();
131            let num_constraints = cs.num_constraints();
132            a[start..end].copy_from_slice(&u[(start + num_constraints)..(end + num_constraints)]);
133        }
134
135        for (i, u_i) in u.iter().enumerate().take(cs.num_constraints()) {
136            for &(ref coeff, index) in &matrices.a[i] {
137                a[index] += &(*u_i * coeff);
138            }
139            for &(ref coeff, index) in &matrices.b[i] {
140                b[index] += &(*u_i * coeff);
141            }
142            for &(ref coeff, index) in &matrices.c[i] {
143                c[index] += &(*u_i * coeff);
144            }
145        }
146
147        Ok((a, b, c, zt, qap_num_variables, domain_size))
148    }
149
150    fn witness_map_from_matrices<F: PrimeField, D: EvaluationDomain<F>>(
151        matrices: &ConstraintMatrices<F>,
152        num_inputs: usize,
153        num_constraints: usize,
154        full_assignment: &[F],
155    ) -> R1CSResult<Vec<F>> {
156        let domain =
157            D::new(num_constraints + num_inputs).ok_or(SynthesisError::PolynomialDegreeTooLarge)?;
158        let domain_size = domain.size();
159        let zero = F::zero();
160
161        let mut a = vec![zero; domain_size];
162        let mut b = vec![zero; domain_size];
163
164        cfg_iter_mut!(a[..num_constraints])
165            .zip(cfg_iter_mut!(b[..num_constraints]))
166            .zip(cfg_iter!(&matrices.a))
167            .zip(cfg_iter!(&matrices.b))
168            .for_each(|(((a, b), at_i), bt_i)| {
169                *a = evaluate_constraint(&at_i, &full_assignment);
170                *b = evaluate_constraint(&bt_i, &full_assignment);
171            });
172
173        {
174            let start = num_constraints;
175            let end = start + num_inputs;
176            a[start..end].clone_from_slice(&full_assignment[..num_inputs]);
177        }
178
179        domain.ifft_in_place(&mut a);
180        domain.ifft_in_place(&mut b);
181
182        let coset_domain = domain.get_coset(F::GENERATOR).unwrap();
183
184        coset_domain.fft_in_place(&mut a);
185        coset_domain.fft_in_place(&mut b);
186
187        let mut ab = domain.mul_polynomials_in_evaluation_domain(&a, &b);
188        drop(a);
189        drop(b);
190
191        let mut c = vec![zero; domain_size];
192        cfg_iter_mut!(c[..num_constraints])
193            .enumerate()
194            .for_each(|(i, c)| {
195                *c = evaluate_constraint(&matrices.c[i], &full_assignment);
196            });
197
198        domain.ifft_in_place(&mut c);
199        coset_domain.fft_in_place(&mut c);
200
201        let vanishing_polynomial_over_coset = domain
202            .evaluate_vanishing_polynomial(F::GENERATOR)
203            .inverse()
204            .unwrap();
205        cfg_iter_mut!(ab).zip(c).for_each(|(ab_i, c_i)| {
206            *ab_i -= &c_i;
207            *ab_i *= &vanishing_polynomial_over_coset;
208        });
209
210        coset_domain.ifft_in_place(&mut ab);
211
212        Ok(ab)
213    }
214
215    fn h_query_scalars<F: PrimeField, D: EvaluationDomain<F>>(
216        max_power: usize,
217        t: F,
218        zt: F,
219        delta_inverse: F,
220    ) -> Result<Vec<F>, SynthesisError> {
221        let scalars = cfg_into_iter!(0..max_power)
222            .map(|i| zt * &delta_inverse * &t.pow([i as u64]))
223            .collect::<Vec<_>>();
224        Ok(scalars)
225    }
226}
